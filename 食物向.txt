    // 辅助函数：获取核心三个选项的组合
    function getCoreCombination() {
        const q1 = document.querySelector('input[name="q1"]:checked');
        const q2 = document.querySelector('input[name="q2"]:checked');
        const q3 = document.querySelector('input[name="q3"]:checked');
        if (!q1 || !q2 || !q3) return null;
        return q1.value + q2.value + q3.value;
    }

    // 获取额外选择的动物与口味文案
    function getExtraMessages() {
        const animalRadio = document.querySelector('input[name="extraAnimal"]:checked');
        const tasteRadio = document.querySelector('input[name="extraTaste"]:checked');
        let animalMsg = "";
        if (animalRadio) {
            const animalVal = animalRadio.value;
            animalMsg = `🐾 你的灵魂小动物是 ${animalVal}，它让这份食物多了一丝俏皮的灵气～ ✨`;
        } else {
            animalMsg = "🐣 没有选小动物也没关系，你的可爱自成一派！";
        }
        let tasteMsg = "";
        if (tasteRadio) {
            const tasteVal = tasteRadio.value;
            tasteMsg = `${tasteVal} 和你的专属食物碰撞出了超治愈的小火花～ 🌈`;
        } else {
            tasteMsg = "🍬 无论什么口味，你都是独特又美好的存在！";
        }
        return { animalMsg, tasteMsg };
    }

    // 显示结果
    function showResult() {
        const combo = getCoreCombination();
        if (!combo) {
            alert("🥺 请先完成前面三个核心问题，才能召唤你的专属食物哟～");
            return;
        }

        let foodName = foodData[combo];
        if (!foodName) foodName = "幸运小汤圆"; // fallback
        const foodInfo = descriptions[foodName] || descriptions["幸运小汤圆"];
        const foodEmoji = foodInfo.emoji;
        const foodDesc = foodInfo.text;

        // 更新结果区
        document.getElementById("resultEmoji").innerText = foodEmoji;
        document.getElementById("resultFoodName").innerText = foodName;
        document.getElementById("resultDesc").innerHTML = `🍽️ ${foodDesc}`;

        // 额外彩蛋信息
        const { animalMsg, tasteMsg } = getExtraMessages();
        const extraHtml = `<div style="margin-bottom: 8px;">🌸 ${animalMsg}</div><div>🍬 ${tasteMsg}</div>`;
        document.getElementById("extraMessage").innerHTML = extraHtml;

        // 显示结果区域
        const resultDiv = document.getElementById("resultBox");
        resultDiv.style.display = "block";
        // 滚动到结果
        resultDiv.scrollIntoView({ behavior: "smooth", block: "nearest" });
    }

    // 分享功能（复制到剪贴板）
    function shareResult() {
        const combo = getCoreCombination();
        if (!combo) {
            alert("先测出你的食物再分享吧～ 🥰");
            return;
        }
        let foodName = foodData[combo];
        if (!foodName) foodName = "幸运小汤圆";
        const foodInfo = descriptions[foodName] || descriptions["幸运小汤圆"];
        const shareText = `🍥 我刚刚测出自己是「${foodName}」${foodInfo.emoji}！ ✨ 超治愈的食物人格~ 你也来测测看你是哪种可爱小吃货？ 🥟🍜 #治愈系食物测试`;
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(shareText).then(() => {
                alert("✨ 复制成功！快去分享给朋友看看你的食物人格~");
            }).catch(() => {
                alert("📋 手动复制一下叭：" + shareText);
            });
        } else {
            alert("📋 复制这段文字分享吧：" + shareText);
        }
    }

    // 绑定提交和分享按钮
    document.getElementById("submitBtn").addEventListener("click", showResult);
    document.getElementById("shareResultBtn").addEventListener("click", shareResult);

    // 让选项卡片更友好（点击整块卡片选中radio）
    const allOptionDivs = document.querySelectorAll('.option');
    allOptionDivs.forEach(div => {
        div.addEventListener('click', (e) => {
            // 防止冒泡到内部label造成重复
            if (e.target.tagName === 'INPUT') return;
            const radio = div.querySelector('input[type="radio"]');
            if (radio) {
                radio.checked = true;
                // 触发一个小反馈
                div.style.transform = "scale(0.98)";
                setTimeout(() => { div.style.transform = ""; }, 100);
            }
        });
    });
